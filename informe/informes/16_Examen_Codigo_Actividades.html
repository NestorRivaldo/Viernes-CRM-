SELECT a.ID_ACTIVIDAD, a.TITULO, a.TIPO, a.ESTADO, a.FECHA,
       c.NOMBRE AS CLIENTE,
       l.NOMBRE AS LEAD,
       o.NOMBRE AS OPORTUNIDAD
FROM CRM_ACTIVIDADES a
LEFT JOIN CRM_CLIENTES c      ON c.ID = a.ID_CLIENTE
